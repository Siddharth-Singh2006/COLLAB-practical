package com.example.demo;
import org.springframework.web.bind.annotation.*;
import java.util.*;
@RestController
@RequestMapping("/api")
public class StudentController {
List<String> students = new ArrayList<>();
// GET API
@GetMapping("/students")
public List<String> getStudents() {
return students;
}
// POST API
@PostMapping("/students")
public String addStudent(@RequestBody String name) {
students.add(name);
return "Student added successfully";
}
// PUT API
@PutMapping("/students/{index}")
public String updateStudent(@PathVariable int index, @RequestBody String name) {
students.set(index, name);
return "Student updated successfully";
}
// DELETE API
@DeleteMapping("/students/{index}")
public String deleteStudent(@PathVariable int index) {
students.remove(index);
return "Student deleted successfully";
}
}

class Calculator {
int add(int a, int b) {
return a + b;
}
double add(double a, double b) {
return a + b;
}
int add(int a, int b, int c) {
return a + b + c;
}
}
public class MethodOverloadingDemo {
public static void main(String[] args) {
Calculator calc = new Calculator();
System.out.println(calc.add(10, 20));

System.out.println(calc.add(5, 10, 15));
}
}

package com.example.demo.controller;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;
import com.example.demo.model.SuchetaUser;
import com.example.demo.service.SuchetaUserService;
import java.util.List;
@CrossOrigin(origins = "*")
@RestController
@RequestMapping("/api/users")
public class SuchetaController
{
@Autowired
private SuchetaUserService service;
@GetMapping
public List<SuchetaUser> getUsers()
{
return service.getAllUsers();
}
@PostMapping
public SuchetaUser addUser(@RequestBody SuchetaUser user)
{
return service.saveUser(user);
}
@DeleteMapping("/{id}")
public String deleteUser(@PathVariable Long id)
{
service.deleteUser(id);
return "User Deleted Successfully";
}
}

function getUsers() {
axios.get("http://localhost:8080/api/users")
.then(res => {
console.log(res.data);
let list = document.getElementById("userList");
list.innerHTML = "";
res.data.forEach(u => {
let li = document.createElement("li");
li.innerText = u.name + " - " + u.email;
list.appendChild(li);
});
})
.catch(err => console.error(err));
}
function addUser() {
let name = document.getElementById("name").value;
let email = document.getElementById("email").value;

email: email
})
.then(() => {

})
.catch(err => console.error(err));
